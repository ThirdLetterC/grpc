# ulog is a source-only C23 dependency, so populate it through FetchContent and
# provide the small library target it does not define itself. Pin the revision
# to keep configure-time downloads reproducible.
if(POLICY CMP0169)
  cmake_policy(SET CMP0169 OLD)
endif()
FetchContent_Declare(ulog
  GIT_REPOSITORY https://github.com/ThirdLetterC/ulog.git
  GIT_TAG 99cf8b6e515eee0353eddd024b354551fd99f2dc
  GIT_PROGRESS TRUE)
FetchContent_GetProperties(ulog)
if(NOT ulog_POPULATED)
  FetchContent_Populate(ulog)
endif()
set(ULOG_SOURCE "${ulog_SOURCE_DIR}/src/ulog.c")
if(NOT GRPC_HAS_C23_CONSTEXPR)
  # Older C2x compilers cannot use const objects as file-scope array bounds.
  # Restore integer constant expressions for the two ulog arrays that need them.
  file(READ "${ULOG_SOURCE}" ulog_source)
  string(REPLACE "static constexpr size_t level_name_storage_size = 64U;"
                 "enum { level_name_storage_size = 64U };" ulog_source "${ulog_source}")
  string(REPLACE "static constexpr int output_total_num =\n    (1 + ULOG_BUILD_EXTRA_OUTPUTS);  // stdout + extra"
                 "enum { output_total_num = (1 + ULOG_BUILD_EXTRA_OUTPUTS) };" ulog_source "${ulog_source}")
  string(REPLACE "static constexpr int output_total_num = 1;  // Only stdout"
                 "enum { output_total_num = 1 };" ulog_source "${ulog_source}")
  set(ulog_compat_dir "${CMAKE_CURRENT_BINARY_DIR}/generated/ulog")
  file(MAKE_DIRECTORY "${ulog_compat_dir}")
  set(ULOG_SOURCE "${ulog_compat_dir}/ulog.c")
  file(WRITE "${ULOG_SOURCE}" "${ulog_source}")
endif()
add_library(ulog STATIC "${ULOG_SOURCE}")

# ulog also supports pre-C23 GCC/Clang through __auto_type. Use that spelling
# for compilers that only accept the draft -std=c2x mode.

  target_compile_options(ulog PRIVATE -include stdbool.h)

endif()

  target_compile_definitions(ulog PRIVATE "nullptr=((void*)0)")
endif()

function(grpc_feature_enabled feature found description)
  string(TOUPPER "${GRPC_${feature}}" setting)
  if(setting STREQUAL "ON" AND NOT found)
    message(FATAL_ERROR "${description} was requested but its dependency was not found")
  endif()
  if(setting STREQUAL "OFF")
    set(enabled FALSE)
  elseif(found)
    set(enabled TRUE)
  else()
    set(enabled FALSE)
  endif()
  set(GRPC_${feature}_ENABLED "${enabled}" PARENT_SCOPE)
endfunction()

    find_program(GRPC_HOST_OPENSSL_EXECUTABLE NAMES openssl REQUIRED NO_CMAKE_PATH)
    set(tls_test_dir "${CMAKE_CURRENT_BINARY_DIR}/generated/tls-tests")
    file(MAKE_DIRECTORY "${tls_test_dir}")
    add_custom_command(OUTPUT "${tls_test_dir}/test_cert.pem" "${tls_test_dir}/test_key.pem"
      COMMAND "${GRPC_HOST_OPENSSL_EXECUTABLE}" req -batch -x509 -newkey rsa:2048 -nodes -keyout "${tls_test_dir}/test_key.pem" -out "${tls_test_dir}/test_cert.pem" -days 365 -subj /CN=localhost
      VERBATIM)
    add_custom_target(test_tls_certificates DEPENDS "${tls_test_dir}/test_cert.pem" "${tls_test_dir}/test_key.pem")
    add_executable(test_tls src/tests/tls.c src/tests/unity/unity.c)
    target_include_directories(test_tls PRIVATE src/tests/unity include/internal)
    if(NGHTTP2_HEADER_DIR)
      target_include_directories(test_tls PRIVATE "${NGHTTP2_HEADER_DIR}")
    endif()
    target_compile_definitions(test_tls PRIVATE "EXPECTED_TLS_BACKEND=\"${_tls_backend}\"")
    target_link_libraries(test_tls PRIVATE grpc_static)
    add_dependencies(test_tls test_tls_certificates)
    add_test(NAME tls COMMAND test_tls)
    set_tests_properties(tls PROPERTIES WORKING_DIRECTORY "${tls_test_dir}")
    list(APPEND GRPC_TEST_TARGETS test_tls)
  endif()

  if(GRPC_ASAN)
    foreach(target IN LISTS GRPC_TEST_TARGETS)
      target_link_libraries(${target} PRIVATE grpc_sanitizers)
    endforeach()
  endif()
endif()
